{ 參數: MINBARS 4  MINAREA 0.15  MINDEPTH 0.04 }

DIF:=EMA(CLOSE,12)-EMA(CLOSE,26);
DEA:=EMA(DIF,9);
HIST:=DIF-DEA;
HP:=HIST/CLOSE*100;
NEG:=HIST<0;
POS:=HIST>0;
NRUN:=IF(NEG,BARSLAST(POS),IF(POS,BARSLAST(NEG),0));
AREA:=SUM(ABS(HP),NRUN);
DEPTH:=HHV(ABS(HP),NRUN);
{ 固定窗口備用: AREA:=SUM(ABS(HP),30); DEPTH:=HHV(ABS(HP),30); }
PBARS:=REF(NRUN,1);  PAREA:=REF(AREA,1);  PDEPTH:=REF(DEPTH,1);
PNEG:=REF(NEG,1);    PPOS:=REF(POS,1);
QUAL:=PBARS>=MINBARS AND PAREA>=MINAREA AND PDEPTH>=MINDEPTH;
BUY:=CROSS(DIF,DEA) AND PNEG AND QUAL;
SELL:=CROSS(DEA,DIF) AND PPOS AND QUAL;

MACD_H:HIST,COLORSTICK;
DIF_L:DIF,COLORFF8800,LINETHICK2;
DEA_L:DEA,COLOR00AAFF;
STICKLINE(BUY,0,DEPTH*CLOSE/100*-1.5,3,0),COLOR00AAFF;
STICKLINE(SELL,0,DEPTH*CLOSE/100*1.5,3,0),COLORFF3333;
DRAWTEXT(BUY,DIF,'B'),COLOR00AAFF;
DRAWTEXT(SELL,DIF,'S'),COLORFF3333;
